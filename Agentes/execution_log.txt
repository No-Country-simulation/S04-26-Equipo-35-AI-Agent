│  def validate_date_format(v):                                                │
│      """Validador personalizado para convertir strings de fecha a objetos    │
│  datetime."""                                                                │
│      if isinstance(v, datetime):                                             │
│          return v                                                            │
│      if isinstance(v, str):                                                  │
│          try:                                                                │
│              # Maneja el formato ISO 8601 con 'Z' para UTC                   │
│              return datetime.fromisoformat(v.replace('Z', '+00:00'))         │
│          except (ValueError, TypeError):                                     │
│              pass                                                            │
│      raise ValueError(f"Formato de fecha inválido: '{v}'. Se esperaba        │
│  formato ISO 8601.")                                                         │

│  # --- 3. Funciones de Limpieza y Normalización ---                          │
│                                                                              │
│  # Patrones de expresiones regulares para limpieza                           │
│  HTML_TAG_PATTERN = re.compile(r'<.*?>')                                     │
│  URL_PATTERN = re.compile(r'https?://\S+|www\.\S+')                          │
│  TIMESTAMP_PATTERN = re.compile(r'\[\d{2}:\d{2}\]')                          │
│  SESSION_ID_PATTERN = re.compile(r'#SES-\d{5}') # Ajustar si el formato de   │
│  ID de sesión varía                                                          │
│  MULTIPLE_SPACES_PATTERN = re.compile(r'\s+')                                │
│  REPEATED_CHARS_PATTERN = re.compile(r'(.)\1{2,}') # Reduce 'aaa' a 'a'      │
│                                                                              │
│  # Emojis de frustración específicos                                         │
│  FRUSTRATION_EMOJIS = {'😤', '😡'}                                           │
│                                                                              │
│  def clean_text(text: str) -> str:                                           │
│      """                                                                     │
│      Aplica una serie de pasos de limpieza y normalización al texto.         │
│      """                                                                     │
│      if not isinstance(text, str):                                           │
│          return ""                                                           │
│                                                                              │
│      # 1. Eliminar etiquetas HTML                                            │
│      text = HTML_TAG_PATTERN.sub('', text)                                   │
│                                                                              │
│      # 2. Reemplazar URLs con [URL]                                          │
│      text = URL_PATTERN.sub('[URL]', text)                                   │
│                                                                              │
│      # 3. Eliminar timestamps embebidos                                      │
│      text = TIMESTAMP_PATTERN.sub('', text)                                  │
│                                                                              │
│      # 4. Eliminar IDs de sesión                                             │
│      text = SESSION_ID_PATTERN.sub('', text)                                 │
│                                                                              │
│      # 5. Manejo de emojis: frustración -> [EMOJI_FRUSTRADO], otros ->       │
│  eliminar                                                                    │
│      cleaned_emojis_text = []                                                │
│      for char in text:                                                       │
│          if emoji.is_emoji(char):                                            │
│              if char in FRUSTRATION_EMOJIS:                                  │
│                  cleaned_emojis_text.append('[EMOJI_FRUSTRADO]')             │
│              # Los emojis no informativos se eliminan al no ser añadidos     │
│          else:                                                               │
│              cleaned_emojis_text.append(char)                                │
│      text = "".join(cleaned_emojis_text)                                     │
│                                                                              │
│      # 6. Convertir a minúsculas                                             │
│      text = text.lower()                                                     │
│                                                                              │
│      # 7. Normalizar espacios múltiples a un solo espacio y eliminar         │
│  espacios al inicio/final                                                    │
│      text = MULTIPLE_SPACES_PATTERN.sub(' ', text).strip()                   │
│                                                                              │
│      # 8. Reducir caracteres repetidos (ej. "holaaaaa" -> "hola")            │
│      text = REPEATED_CHARS_PATTERN.sub(r'\1', text)                          │
│                                                                              │
│      return text.strip()                                                     │
